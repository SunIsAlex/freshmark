cos⁡x+cos⁡3x+cos⁡5x+⋯+cos⁡(2n−1)x=12[(z+z3+...+z2n−1)+(z‾+z‾3+...+z‾2n−1)]=12[(z2n−1)−(z‾2n−1)z−z‾]=12[(zn+z‾n)(zn−z‾n)z−z‾]=(zn+z‾n2)(zn−z‾n2)(2iz−z‾)=sin⁡nxcos⁡nxsin⁡x\begin{gathered} \cos x + \cos 3x + \cos 5x + \cdots + \cos(2n-1)x\\ =\frac{1}{2}[(z+z^3+...+z^{2n-1})+(\overline{z}+\overline{z}^3+...+\overline{z}^{2n-1})]\\ =\frac{1}{2}[\frac{(z^{2n}-1)-(\overline{z}^{2n}-1)}{z-\overline{z}}]\\ =\frac{1}{2}[\frac{(z^n+\overline{z}^n)(z^n-\overline{z}^n)}{z-\overline{z}}]\\ =(\frac{z^n+\overline{z}^n}{2})(\frac{z^n-\overline{z}^n}{2})(\frac{2i}{z-\overline{z}})=\frac{\sin nx\cos nx}{\sin x} \end{gathered}
